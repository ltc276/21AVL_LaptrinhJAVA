	public static Customer addCustomer(Customer root,Customer e) {
		if  (root==null) return e;
		if (e.getCustomer_ID()<root.getCustomer_ID()) {
			if (root.left==null) root.left = e;
			else root.left = addCustomer(root.left,e);
		}
		else {
			if (root.right==null) root.right = e;
			else root.right = addCustomer(root.right,e);
		}
		return root;
	}
	public static void readtree(Customer root) {
		if (root!=null) {
			readtree(root.left);
			System.out.println();
			System.out.printf("%-10d%-25s%-10s%-10d",root.getCustomer_ID(),root.getCustomer_Name(),root.getCustomer_Gender(),root.getCustomer_Age());
			readtree(root.right);
		}
	}
	public static Customer findCustomer(Customer root,int ID) {
		if (root==null) return null;
		if (ID==root.getCustomer_ID()) return root;
		else if (ID<root.getCustomer_ID()) return findCustomer(root.left,ID);
		else return findCustomer(root.right,ID);
 	}
	
	public static Customer deleteCustomer(Customer root, int ID) {
		if (root==null) return null;
		if (root.getCustomer_ID()==ID) {
			if (root.left==null) return root.right;
			if (root.right==null) return root.left;
			Customer T = theleftCustomer(root.right);
			Customer L = root.left;
			Customer R = deleteCustomer(root.right,T.getCustomer_ID());
			T.left = L;
			T.right = R;
			return T;
		}
		if (ID<root.getCustomer_ID()) root.left = deleteCustomer(root.left,ID);
		else root.right = deleteCustomer(root.right,ID);
		return root;
	}
	public static Customer theleftCustomer(Customer root) {
		if (root==null) return null;
		if (root.left==null) return root;
		else return theleftCustomer(root.left);
	}
